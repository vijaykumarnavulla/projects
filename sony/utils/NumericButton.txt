import React, { useState, useRef } from 'react';
import { View, Text, TouchableOpacity, StyleSheet, Animated } from 'react-native';
import PropTypes from 'prop-types';

const NumericButton = ({ text, onPress, animated = true }) => {
  const animationValue = useRef(new Animated.Value(0)).current;

  const handleButtonPress = () => {
    if (animated) {
      startAnimation();
    }
    onPress();
  };

  const startAnimation = () => {
    Animated.timing(animationValue, {
      toValue: 1,
      duration: 500,
      useNativeDriver: true,
    }).start(() => {
      animationValue.setValue(0);
    });
  };

  const animatedStyle = {
    transform: [
      {
        scale: animationValue.interpolate({
          inputRange: [0, 1],
          outputRange: [1, 1.2],
        }),
      },
    ],
  };

  const ButtonContainer = animated ? Animated.View : View;

  return (
    <View style={styles.container}>
      <ButtonContainer style={[styles.buttonContainer, animated ? animatedStyle : null]}>
        <TouchableOpacity
          style={styles.button}
          onPress={handleButtonPress}
          activeOpacity={0.7}
        >
          <Text style={styles.buttonText}>{text}</Text>
        </TouchableOpacity>
      </ButtonContainer>
    </View>
  );
};

NumericButton.propTypes = {
  text: PropTypes.string.isRequired,
  onPress: PropTypes.func.isRequired,
  animated: PropTypes.bool,
};

NumericButton.defaultProps = {
  animated: true,
};

const styles = StyleSheet.create({
  container: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
  },
  buttonContainer: {
    backgroundColor: '#f2f2f2',
    borderRadius: 8,
    overflow: 'hidden',
  },
  button: {
    width: 100,
    height: 60,
    justifyContent: 'center',
    alignItems: 'center',
  },
  buttonText: {
    fontSize: 24,
    fontWeight: 'bold',
  },
});

export default NumericButton;

import React from 'react';
import { View, Text, StyleSheet, Alert } from 'react-native';

const App = () => {
  const handlePressLetter = (letter) => {
    Alert.alert(`Pressed: ${letter}`);
  };

  const renderLetters = (text) => {
    // Split the string into individual letters and render them as separate <Text> components
    return text.split('').map((char, index) => (
      <Text
        key={index}
        style={styles.letter}
        onPress={() => handlePressLetter(char)}
      >
        {char}
      </Text>
    ));
  };

  return (
    <View style={styles.container}>
      <Text style={styles.text}>
        {renderLetters('Hello world')}
      </Text>
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
    padding: 16,
  },
  text: {
    fontSize: 24,
    flexDirection: 'row',
    flexWrap: 'wrap',
  },
  letter: {
    fontSize: 24,
  },
});

export default App;

const renderWords = (text) => {
  // Split the string into words and render them as separate <Text> components
  return text.split(' ').map((word, index) => (
    <Text
      key={index}
      style={styles.word}
      onPress={() => handlePressLetter(word)}
    >
      {word + ' '}
    </Text>
  ));
};
